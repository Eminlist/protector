/* ============================================================= CHAT İDARƏETMƏSİ ============================================================= */ function openChat() { document.getElementById("chat-view").classList.add("active"); document.body.style.overflow = "hidden"; // Arxanın hərəkət etməsini dayandır triggerHaptic(20); } function closeChat() { document.getElementById("chat-view").classList.remove("active"); document.body.style.overflow = ""; triggerHaptic(15); } function sendChatMessage() { const input = document.getElementById("chatInput"); const messageText = input.value.trim(); if (!messageText) return; const messagesContainer = document.getElementById("chat-messages"); // Göndərilən mesajı yaradırıq const sentBubble = document.createElement("div"); sentBubble.className = "chat-bubble sent"; sentBubble.innerText = messageText; messagesContainer.appendChild(sentBubble); input.value = ""; messagesContainer.scrollTop = messagesContainer.scrollHeight; // Səhifəni ən aşağı sürüşdür playSoundEffect(); // Göndərmə səsi (mövcud səs funksiyası istifadə edilir) // Avtomatik cavab (Yalnız simulyasiya üçündür, bazaya/serverə bağlaya bilərsiniz) setTimeout(() => { const receivedBubble = document.createElement("div"); receivedBubble.className = "chat-bubble received"; receivedBubble.innerText = "Sizin mesajınız qeydə alındı: " + messageText; messagesContainer.appendChild(receivedBubble); messagesContainer.scrollTop = messagesContainer.scrollHeight; triggerHaptic(30); }, 1000); } // "Enter" düyməsinə basdıqda mesajı göndərmək function handleChatKey(event) { if (event.key === "Enter") { sendChatMessage(); } }
